@freezed
abstract class PlayerDto with _$PlayerDto {
  const factory PlayerDto(String id) = _PlayerDto;

  factory PlayerDto.fromJson(Map<String, dynamic> json) =>
      _$PlayerDtoFromJson(json);

  factory PlayerDto.fromPlayer(Player player) => PlayerDto(player.id);
}

@freezed
abstract class PlayerDatabase with _$PlayerDatabase {
  const factory PlayerDatabase(Iterable<Player> players) = _PlayerDatabase;

  factory PlayerDatabase.fromJson(Map<String, dynamic> json) =>
      _$PlayerDatabaseFromJson(json);
}

class PlayerRepository extends InMemoryRepository<Player> {
  PlayerRepository._();

  static final PlayerRepository instance = PlayerRepository._();

  late Map<String, Player> _cache;

  Future<void> initialize() async {
    final db = RepositoryService.instance.db;
    final rows = await db.query('players');
    _cache = Map.fromEntries(
      rows.map((row) => MapEntry(row['id'] as String, _mapToPlayer(row))),
    );
    log('PlayerRepository loaded ${_cache.length} players from SQLite');
  }

  Iterable<Player> getAllCachedPlayers() {
    return _cache.values.toSet();
  }

  @override
  Future<void> add(Player player) async {
    await RepositoryService.instance.db.insert(
      'players',
      _playerToMap(player),
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
    _cache[player.id] = player;
  }

  Future<void> addAll(Iterable<Player> players) async {
    final db = RepositoryService.instance.db;
    await db.transaction((txn) async {
      for (final player in players) {
        await txn.insert(
          'players',
          _playerToMap(player),
          conflictAlgorithm: ConflictAlgorithm.replace,
        );
      }
    });
    _cache.addAll(Map.fromEntries(players.map((e) => MapEntry(e.id, e))));
  }

  @override
  Future<void> delete(String id) async {
    await RepositoryService.instance.db.delete(
      'players',
      where: 'id = ?',
      whereArgs: [id],
    );
    _cache.remove(id);
  }

  @override
  Future<Player?> get(String id) async {
    return _cache[id];
  }

  @override
  Future<Iterable<Player>> getAll({int? count, int? offset}) async {
    final skippedEntries = _cache.values.skip(offset ?? 0);
    if (count != null) {
      return skippedEntries.take(count);
    }
    return skippedEntries;
  }

  Future<Player?> getPlayerFromDto(PlayerDto dto) async {
    return _cache[dto.id];
  }

  @override
  Future<void> update(String id, Player value) async {
    await RepositoryService.instance.db.update(
      'players',
      _playerToMap(value),
      where: 'id = ?',
      whereArgs: [id],
    );
    _cache.update(id, (_) => value);
  }

  Map<String, dynamic> _playerToMap(Player p) => {
    'id': p.id,
    'first_name': p.firstName,
    'last_name': p.lastName,
    'elo': p.elo.name,
    'sign_in_id': p.signInId,
    'created_date': p.createdDate?.toIso8601String(),
  };

  Player _mapToPlayer(Map<String, dynamic> row) => Player(
    id: row['id'] as String,
    firstName: row['first_name'] as String,
    lastName: row['last_name'] as String,
    elo: Elo.values.firstWhere((e) => e.name == row['elo']),
    signInId: row['sign_in_id'] as String?,
    createdDate: row['created_date'] != null
        ? DateTime.parse(row['created_date'] as String)
        : null,
  );
}

import 'package:freezed_annotation/freezed_annotation.dart';
import 'package:good_bad_board/model/game.dart';
import 'package:good_bad_board/repository/player_repository.dart';

import 'package:good_bad_board/repository/repository_service.dart';
import 'package:sqflite/sqflite.dart';

part 'game_repository.freezed.dart';
part 'game_repository.g.dart';

@freezed
abstract class GameDto with _$GameDto {
  const factory GameDto(
    Iterable<PlayerDto> playerDto,
    String id,
    DateTime createdDate,
  ) = _GameDto;

  factory GameDto.fromJson(Map<String, dynamic> json) =>
      _$GameDtoFromJson(json);

  factory GameDto.fromGame(Game game) => GameDto(
    game.players.map(PlayerDto.fromPlayer),
    game.id,
    game.createdDate,
  );
}

@freezed
abstract class GameDatabase with _$GameDatabase {
  const factory GameDatabase(Iterable<GameDto> games) = _GameDatabase;

  factory GameDatabase.fromJson(Map<String, dynamic> json) =>
      _$GameDatabaseFromJson(json);
}

class GameRepository extends InMemoryRepository<Game> {
  GameRepository._();

  static final GameRepository instance = GameRepository._();

  late Map<String, Game> _cache;

  late PlayerRepository _playerRepository;

  Future<void> initialize() async {
    _playerRepository = PlayerRepository.instance;
    final db = RepositoryService.instance.db;

    final gameRows = await db.query('games');
    final games = await Future.wait(
      gameRows.map((row) async {
        final playerRows = await db.query(
          'game_players',
          where: 'game_id = ?',
          whereArgs: [row['id']],
          orderBy: 'position ASC',
        );
        final maybePlayers = await Future.wait(
          playerRows.map(
            (pr) => _playerRepository.get(pr['player_id'] as String),
          ),
        );
        return Game(
          players: maybePlayers.nonNulls.toList(growable: true),
          id: row['id'] as String,
          createdDate: DateTime.parse(row['created_date'] as String),
        );
      }),
    );
    _cache = Map.fromEntries(games.map((g) => MapEntry(g.id, g)));
    log('GameRepository loaded ${_cache.length} games from SQLite');
  }

  @override
  Future<void> add(Game game) async {
    final db = RepositoryService.instance.db;
    await db.transaction((txn) async {
      await txn.insert(
        'games',
        {'id': game.id, 'created_date': game.createdDate.toIso8601String()},
        conflictAlgorithm: ConflictAlgorithm.replace,
      );
      for (int i = 0; i < game.players.length; i++) {
        await txn.insert(
          'game_players',
          {
            'game_id': game.id,
            'player_id': game.players[i].id,
            'position': i,
          },
          conflictAlgorithm: ConflictAlgorithm.replace,
        );
      }
    });
    _cache[game.id] = game;
  }

  Future<void> addAll(Iterable<Game> games) async {
    for (final game in games) {
      await add(game);
    }
  }

  @override
  Future<void> delete(String id) async {
    final db = RepositoryService.instance.db;
    await db.transaction((txn) async {
      await txn.delete(
        'game_players',
        where: 'game_id = ?',
        whereArgs: [id],
      );
      await txn.delete(
        'games',
        where: 'id = ?',
        whereArgs: [id],
      );
    });
    _cache.remove(id);
  }

  @override
  Future<Game?> get(String id) async {
    return _cache[id];
  }

  @override
  Future<Iterable<Game>> getAll({int? count, int? offset}) async {
    final skippedEntries = _cache.values.skip(offset ?? 0);
    if (count != null) {
      return skippedEntries.take(count);
    }
    return skippedEntries;
  }

  @override
  Future<void> update(String id, Game value) async {
    final db = RepositoryService.instance.db;
    await db.transaction((txn) async {
      await txn.update(
        'games',
        {'created_date': value.createdDate.toIso8601String()},
        where: 'id = ?',
        whereArgs: [id],
      );
      await txn.delete(
        'game_players',
        where: 'game_id = ?',
        whereArgs: [id],
      );
      for (int i = 0; i < value.players.length; i++) {
        await txn.insert(
          'game_players',
          {'game_id': id, 'player_id': value.players[i].id, 'position': i},
          conflictAlgorithm: ConflictAlgorithm.replace,
        );
      }
    });
    _cache.update(id, (_) => value);
  }
}
